SQL Features Demonstrated in Project (Viva Reference) 

-- 1. SELECT: Retrieve all available Deluxe rooms
SELECT * FROM rooms WHERE status = 'Available' AND room_type = 'Deluxe';

-- 2. INSERT: Add a new room with constraints
INSERT INTO rooms (room_number, room_type, capacity, price_per_night, amenities, status)
VALUES ('601', 'Suite', 3, 5500.00, 'Wi-Fi, Jacuzzi, Balcony', 'Available');

-- 3. UPDATE: Update guest contact info
UPDATE users SET phone = '+91 9999999999' WHERE user_id = 1002;

-- 4. DELETE: Remove unused room
DELETE FROM rooms WHERE room_id = 999;

-- INNER JOIN: Bookings with User and Room entities
SELECT b.booking_id, u.full_name, u.email, r.room_number, r.room_type, b.check_in, b.check_out, b.total_amount
FROM bookings b
INNER JOIN users u ON b.user_id = u.user_id
INNER JOIN rooms r ON b.room_id = r.room_id
WHERE b.booking_status = 'Confirmed';

-- LEFT JOIN: Room category performance (including categories with zero bookings)
SELECT r.room_type, COUNT(b.booking_id) AS total_bookings, COALESCE(SUM(b.total_amount), 0) AS revenue
FROM rooms r
LEFT JOIN bookings b ON r.room_id = b.room_id
GROUP BY r.room_type;

-- Correlated subquery using NOT EXISTS for date collision detection:
SELECT r.* FROM rooms r
WHERE r.status = 'Available'
AND NOT EXISTS (
    SELECT 1 FROM bookings b
    WHERE b.room_id = r.room_id
      AND b.booking_status IN ('Confirmed', 'Checked In')
      AND b.check_in < '2026-10-25' AND b.check_out > '2026-10-20'
);

-- Subquery in WHERE clause with IN:
SELECT full_name, email FROM users
WHERE user_id IN (
    SELECT DISTINCT user_id FROM bookings WHERE total_amount > 10000
);

-- Aggregation: Room pricing and counts by category with HAVING clause
SELECT 
  room_type,
  COUNT(*) AS total_rooms,
  MIN(price_per_night) AS lowest_rate,
  MAX(price_per_night) AS highest_rate,
  ROUND(AVG(price_per_night), 2) AS average_rate
FROM rooms
GROUP BY room_type
HAVING COUNT(*) >= 2;

-- Trigger Definition (Status Audit)
CREATE TRIGGER trg_booking_status_audit
AFTER UPDATE ON bookings FOR EACH ROW
BEGIN
  IF OLD.booking_status <> NEW.booking_status THEN
    INSERT INTO booking_audit (booking_id, action_type, old_status, new_status, details)
    VALUES (NEW.booking_id, 'STATUS_CHANGED', OLD.booking_status, NEW.booking_status, 'Updated');
  END IF;
END;
